  141| def comoving_distance(z_arr, n_steps=500):

  205| # HOD (identical mean occupation to phase5_hod_b3.py) + velocities (option a)

  207| def mean_Ncen(mass_h, log_Mmin, sigma_logM):
  208|     from scipy.special import erf
  209|     return 0.5 * (1.0 + erf((np.log10(mass_h) - log_Mmin) / (sigma_logM + 1e-10)))

  211| def mean_Nsat(mass_h, log_M0, log_M1, alpha, log_Mmin):
  212|     M0, M1 = 10 ** log_M0, 10 ** log_M1
  213|     N_sat = np.zeros(len(mass_h))
  214|     m = mass_h > M0
  215|     ratio = np.where(m, (mass_h - M0) / (M1 + 1e-30), 0.0)
  216|     N_sat[m] = ratio[m] ** alpha
  217|     N_sat *= mean_Ncen(mass_h, log_Mmin, 0.2)
  218|     return N_sat

  220| def populate_halos_hod_with_vel(pos_h, mass_h, vel_h, hod_params, rng):
  221|     """Return (pos_gal [N,3] Mpc/h, vel_gal [N,3] km/s). Option (a): satellites
  222|     inherit the parent halo bulk velocity, no intra-halo dispersion."""
  223|     log_Mmin, sigma_logM, log_M0, log_M1, alpha, A_cen, A_sat, eta_vel, eta_conc = hod_params
  224|     N_h = len(mass_h)
  225|     if N_h == 0:
  226|         return np.zeros((0, 3)), np.zeros((0, 3))
  227| 
  228|     p_cen = np.clip(mean_Ncen(mass_h, log_Mmin, sigma_logM), 0.0, 1.0)
  229|     is_central = rng.random(N_h) < p_cen
  230|     lam_sat = np.clip(mean_Nsat(mass_h, log_M0, log_M1, alpha, log_Mmin), 0.0, 1e4)
  231|     n_sat = rng.poisson(lam_sat)

  233|     pos_list, vel_list = [], []
  234|     if is_central.any():
  235|         pos_list.append(pos_h[is_central])
  236|         vel_list.append(vel_h[is_central])

  238|     rho_crit = 2.775e11 * OMM

  240|         ns = int(n_sat[i])
  241|         if ns <= 0:
  242|             continue
  243|         r_vir = np.clip((3.0 * mass_h[i] / (4.0 * np.pi * 200.0 * rho_crit)) ** (1.0 / 3.0),
  244|                         0.01, 5.0) * eta_conc

  246|         r = r_vir * u ** (1.0 / 3.0)
  247|         theta = np.arccos(1.0 - 2.0 * rng.random(ns))
  248|         phi = 2.0 * np.pi * rng.random(ns)

  250|                                 r * np.sin(theta) * np.sin(phi),
  251|                                 r * np.cos(theta)])
  252|         pos_list.append((pos_h[i] + dxyz) % BOXSIZE_MOCK)

  261| # CIC (identical to phase6_bgs_voxelize.cic_3d)
  262| # ---------------------------------------------------------------------------
  263| def cic_3d(pos, weights, ngrid, box_min, box_size):
  264|     cell = box_size / ngrid
  265|     xyz  = np.clip((pos - box_min[None, :]) / cell, 0.0, ngrid - 1e-6)
  266|     ijk  = xyz.astype(np.int32)
  267|     d    = (xyz - ijk).astype(np.float32)
  268|     w    = weights.astype(np.float32)
  269|     flat = np.zeros(ngrid ** 3, dtype=np.float32)
  270|     for di in range(2):
  271|         wx = (1.0 - d[:, 0]) if di == 0 else d[:, 0]
  272|         ii = np.clip(ijk[:, 0] + di, 0, ngrid - 1)
  273|         for dj in range(2):
  274|             wy = (1.0 - d[:, 1]) if dj == 0 else d[:, 1]
  275|             jj = np.clip(ijk[:, 1] + dj, 0, ngrid - 1)
  276|             for dk in range(2):
  277|                 wz = (1.0 - d[:, 2]) if dk == 0 else d[:, 2]
  278|                 kk = np.clip(ijk[:, 2] + dk, 0, ngrid - 1)
  279|                 idx = ii * ngrid ** 2 + jj * ngrid + kk
  280|                 flat += np.bincount(idx, weights=wx * wy * wz * w,
  281|                                     minlength=ngrid ** 3).astype(np.float32)
  282|     return flat.reshape(ngrid, ngrid, ngrid)

  285| # ---------------------------------------------------------------------------
  286| # UNMASKED TDA — byte-for-byte the DESI canonical (phase6_bgs_tda.compute_tda_features)
  287| # The CubicalComplex runs on the FULL -field (exterior zeros included). mask is
  288| # used only for thresholds and the b0 mean level — identical to DESI.
  289| # ---------------------------------------------------------------------------
  290| def compute_tda_features(delta_field, mask, n_thresh=100, masked=False):
  291|     import gudhi
  292|     # The log-transform is applied upstream in build_field (on the RAW delta).
  293|     # compute_tda_features works on the field as-is.
  294|     #
  295|     # masked=False (Test 1): CubicalComplex on the FULL -field, exterior zeros
  296|     #   included in the filtration — identical to the (broken) DESI canonical.
  297|     # masked=True (Test 2): exterior EXCLUDED from the filtration. Exterior cells
  298|     #   are set to a high sentinel in -field so they enter the sublevel filtration
  299|     #   LAST; any H1 loop that would need the exterior to close is cut (its birth
  300|     #   or death touches the sentinel and is removed). Applied identically to DESI
  301|     #   and mocks — the only correct like-for-like treatment for a bounded survey.
  302|     field = delta_field.astype(np.float64)
  303|     field_in = field[mask]
  304|     nu_min = float(np.percentile(field_in, 1))
  305|     nu_max = float(np.percentile(field_in, 99))
  306|     thresholds = np.linspace(nu_min, nu_max, n_thresh)
  307| 
  308|     SENT = 1.0e6
  309|     if masked:
  310|         field_work = field.copy()
  311|         field_work[~mask] = -SENT          # exterior very low in field ...
  312|         field_neg = -field_work            # ... => +SENT in -field (enters last)
  313|         cutoff = SENT / 2.0
  314|     else:
  315|         field_neg = -field
  316|         cutoff = np.inf
  317| 
  318|     cc = gudhi.CubicalComplex(dimensions=list(field_neg.shape),
  319|                               top_dimensional_cells=field_neg.flatten())
  320|     cc.compute_persistence()
  321|     diag_0 = cc.persistence_intervals_in_dimension(0)
  322|     diag_1 = cc.persistence_intervals_in_dimension(1)
  323| 
  324|     def proc(diag):
  325|         if len(diag) == 0:
  326|             return np.array([]), np.array([]), np.array([])
  327|         d = np.array(diag)
  328|         keep = np.isfinite(d[:, 1]) & (d[:, 0] < cutoff) & (d[:, 1] < cutoff)
  329|         df = d[keep]
  330|         b, dth = -df[:, 0], -df[:, 1]
  331|         return b, dth, b - dth
  332| 
  333|     b0, d0, _   = proc(diag_0)
  334|     b1, d1, p1  = proc(diag_1)
  335| 
  336|     b1_curve = np.zeros(n_thresh)
  337|     b0_curve = np.zeros(n_thresh)
  338|     for k, nu in enumerate(thresholds):
  339|         if len(b0):
  340|             b0_curve[k] = np.sum((b0 >= nu) & (d0 < nu))
  341|         if len(b1):
  342|             b1_curve[k] = np.sum((b1 >= nu) & (d1 < nu))
  343| 
  344|     feats = np.zeros(8, dtype=np.float64)
  345|     if b1_curve.max() > 0:
  346|         pk = np.argmax(b1_curve)
  347|         feats[0] = thresholds[pk]
  348|         feats[1] = b1_curve[pk]
  349|         half = b1_curve.max() / 2.0
  350|         above = np.where(b1_curve >= half)[0]
  351|         feats[2] = (thresholds[above[-1]] - thresholds[above[0]]) if len(above) > 1 else 0.0
  352|         feats[3] = np.trapezoid(b1_curve, thresholds)
  353|     if len(p1) > 0:
  354|         feats[4] = len(p1)
  355|         feats[5] = float(np.mean(p1))
  356|         feats[6] = float(np.sum(p1 >= np.percentile(p1, 90)))
  357|     mean_val = float(field[mask].mean())
  358|     feats[7] = b0_curve[np.argmin(np.abs(thresholds - mean_val))]
  359|     # feats[4] = beta1_max (n loops), feats[5] = <pers1>
  360|     return feats

  489| def build_field(field_d, field_r, alpha, mask):
  490|     """Shared field construction for DESI and cut-sky mocks. Identical byte-for-byte
  491|     on both sides. Pipeline (as STATED in the paper):
  492|       raw delta_FKP (>= -1)  ->  nu = log(1 + clip(delta, -1))  ->  smooth  ->  mean-sub.
  493|     The log on the RAW delta bounds the FKP shot-noise positive tail (delta up to
  494|     +150 -> log(151)=5.0, matching DESI's raw range) without a spurious negative
  495|     spike. Exterior/invalid voxels are 0 (= log(1)), preserving the zero exterior."""

  497|     denom = alpha * field_r
  498|     valid = denom > 0
  499|     delta[valid] = (field_d[valid] - denom[valid]) / denom[valid]
  500|     delta[~mask] = 0.0
  501|     nu = np.zeros_like(delta)
  502|     nu[mask] = np.log(1.0 + np.clip(delta[mask], -1.0 + 1e-3, None))
  503|     nu = gaussian_filter(nu, sigma=SIGMA_PX)
